cmake_minimum_required(VERSION 3.18)
project(helmer LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Dependencies: FLINT (>= 3.0), GMP, msolve's libneogb (>= 0.6).
find_library(FLINT_LIB flint REQUIRED)
find_library(GMP_LIB gmp REQUIRED)
find_package(OpenMP)

add_library(helmer_core STATIC
  src/poly.cpp src/gb.cpp src/ideal.cpp src/primes.cpp src/whitney.cpp src/certify.cpp)
option(HELMER_F4_TRACE "Build the pinned backend for experimental modular F4 replay" OFF)
if(HELMER_F4_TRACE)
  include(cmake/TraceBackend.cmake)
  target_link_libraries(helmer_core PRIVATE helmer_trace_backend)
  target_sources(helmer_core PRIVATE src/f4_trace.c)
  target_compile_definitions(helmer_core PRIVATE HELMER_F4_TRACE)
else()
  find_library(NEOGB_LIB neogb REQUIRED)
  target_link_libraries(helmer_core PRIVATE ${NEOGB_LIB})
endif()
target_include_directories(helmer_core PUBLIC include)
target_link_libraries(helmer_core PUBLIC ${FLINT_LIB} ${GMP_LIB} m)
if(OpenMP_C_FOUND)
  target_link_libraries(helmer_core PUBLIC OpenMP::OpenMP_C)
endif()

option(HELMER_PYTHON "Build the Python extension" ON)
if(HELMER_PYTHON)
  find_package(Python 3.9 COMPONENTS Interpreter Development.Module REQUIRED)
  execute_process(COMMAND ${Python_EXECUTABLE} -m pybind11 --cmakedir
                  OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  find_package(pybind11 CONFIG REQUIRED)
  pybind11_add_module(_helmer src/python_bindings.cpp)
  target_link_libraries(_helmer PRIVATE helmer_core)
  install(TARGETS _helmer DESTINATION helmer)
endif()

option(HELMER_TESTS "Build C++ test drivers" ON)
if(HELMER_TESTS)
  foreach(t t_gb t_primes)
    add_executable(${t} tests/${t}.cpp)
    target_link_libraries(${t} PRIVATE helmer_core)
  endforeach()
endif()

if(HELMER_TESTS AND HELMER_F4_TRACE)
  add_executable(t_f4_trace tests/t_f4_trace.cpp)
  target_link_libraries(t_f4_trace PRIVATE helmer_core)
  enable_testing()
  add_test(NAME f4_trace COMMAND t_f4_trace)
  find_program(VALGRIND_EXECUTABLE valgrind)
  if(VALGRIND_EXECUTABLE)
    execute_process(COMMAND ${VALGRIND_EXECUTABLE} --tool=memcheck --help
      OUTPUT_VARIABLE memcheck_help)
    if(memcheck_help MATCHES "show-realloc-size-zero")
      set(memcheck_options --show-realloc-size-zero=no)
    endif()
    add_test(NAME f4_trace_memcheck COMMAND ${VALGRIND_EXECUTABLE}
      ${memcheck_options} --leak-check=full --errors-for-leak-kinds=definite,indirect
      --error-exitcode=1 $<TARGET_FILE:t_f4_trace>)
  endif()
endif()
